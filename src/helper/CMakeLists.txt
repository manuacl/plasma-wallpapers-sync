# Standalone executable installed at $KAUTH_HELPER_INSTALL_DIR
# (typically /usr/libexec/kauth or /usr/libexec depending on distro).
# Polkit's D-Bus service activation invokes it on demand when the
# client (KAuthPrivilegedWriter in this project) calls KAuth::Action.
add_executable(plasma-wallpaper-sync-helper
    PlasmaLoginHelper.cpp
    PlasmaLoginHelper.h
)

target_link_libraries(plasma-wallpaper-sync-helper
    PRIVATE
        Qt6::Core
        Qt6::DBus
        KF6::AuthCore
)

if(NOT KAUTH_HELPER_INSTALL_DIR)
    message(FATAL_ERROR
        "KAUTH_HELPER_INSTALL_DIR is empty — KF6Auth didn't expose it. "
        "Set it manually or upgrade KF6Auth.")
endif()

# Flatpak install (CMAKE_INSTALL_PREFIX=/app) can't write to the
# absolute system paths KAuth uses by default (KAUTH_HELPER_INSTALL_DIR
# = /usr/lib/.../kf6/kauth, KAUTH_POLICY_FILES_INSTALL_DIR =
# /usr/share/polkit-1/actions), and even if it could the host polkit
# daemon wouldn't see the result. Skip those installs under Flatpak —
# the .service + .conf files generated by kauth_install_helper_files
# still land inside the bundle harmlessly. A native install rebuilds
# with CMAKE_INSTALL_PREFIX=/usr where every absolute path lines up.
set(_pws_is_flatpak_install OFF)
if(CMAKE_INSTALL_PREFIX STREQUAL "/app")
    set(_pws_is_flatpak_install ON)
    message(STATUS
        "Flatpak install detected — skipping helper binary + polkit policy "
        "installs. KAuth is not usable from inside a Flatpak sandbox anyway; "
        "see docs for the hybrid host install used during dev testing.")
endif()

# kauth_install_helper_files() only generates and installs the
# support files (.service + .conf) under share/dbus-1/. Installing
# the binary itself is our responsibility.
if(NOT _pws_is_flatpak_install)
    install(TARGETS plasma-wallpaper-sync-helper
        DESTINATION "${KAUTH_HELPER_INSTALL_DIR}"
    )
endif()

kauth_install_helper_files(plasma-wallpaper-sync-helper
    "${APP_ID}"
    "root"
)

# Install the polkit policy XML. The action id inside it
# ("dev.manuacl.plasmawallpapersync.writefile") must match what
# KAuthPrivilegedWriter passes to KAuth::Action.
if(NOT _pws_is_flatpak_install)
    kauth_install_actions(
        "${APP_ID}"
        "${CMAKE_SOURCE_DIR}/data/${APP_ID}.policy"
    )
endif()
